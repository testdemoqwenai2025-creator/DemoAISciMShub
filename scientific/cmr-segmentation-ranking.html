A research deep-dive into Consistency Model Ranking (CMR) by Joshua Talks, Kevin Marchesini, Luca Lumi, Federico Bolelli, and Anna Kreshuk (EMBL Heidelberg · University of Modena). The paper introduces a method to rank segmentation models on unlabeled target data by measuring their prediction consistency under random augmentations — no source labels, no target labels required. This page presents the math, the code (in three frameworks), live animations, and how it complements OSIRIS's Theorem 7 bound.

import torch
import torch.nn.functional as F
import torchvision.transforms as T

def dice_loss(a: torch.Tensor, b: torch.Tensor) -> float:
    """Dice coefficient between two binary masks (B, H, W)."""
    inter = (a * b).sum(dim=(1, 2))
    return (2 * inter / (a.sum(dim=(1, 2)) + b.sum(dim=(1, 2)) + 1e-8)).mean().item()

class RandomAugment:
    def __init__(self):
        self.transforms = T.Compose([
            T.RandomHorizontalFlip(p=0.5),
            T.RandomRotation(degrees=15),
            T.RandomAffine(degrees=0, translate=(0.05, 0.05)),
        ])
    def __call__(self, x: torch.Tensor) -> torch.Tensor:
        return self.transforms(x)

def cmr_score(model: torch.nn.Module, x: torch.Tensor, n_aug: int = 8) -> float:
    """Monte Carlo CMR score for a single image (C, H, W) tensor."""
    model.eval()
    with torch.no_grad():
        aug = RandomAugment()
        pred_orig = torch.sigmoid(model(x.unsqueeze(0)))  # (1, C, H, W)
        scores = []
        for _ in range(n_aug):
            x_aug = aug(x)
            pred_aug = torch.sigmoid(model(x_aug.unsqueeze(0)))
            # Inverse warp is omitted here for brevity; in practice we
            # store the affine matrix and use F.affine_grid + F.grid_sample
            # with the inverse to back-warp pred_aug to match pred_orig.
            scores.append(dice_loss(pred_orig[0], pred_aug[0]))
        return sum(scores) / len(scores)

def rank_models(models: list, target_loader: torch.utils.data.DataLoader,
                n_aug: int = 8, device: str = 'cuda') -> list:
    """Rank models by CMR score on unlabeled target dataset."""
    results = []
    for i, M in enumerate(models):
        M = M.to(device)
        scores = []
        for batch in target_loader:
            x = batch.to(device)
            scores.append(cmr_score(M, x, n_aug))
        avg = sum(scores) / len(scores)
        results.append({'model': i, 'cmr_score': avg})
    return sorted(results, key=lambda r: -r['cmr_score'])

Experiments — ranking accuracy

The paper reports Spearman rank correlation ρ between CMR's predicted ranking and the ground-truth ranking (by Dice / PQ on labeled target test set). Higher ρ = better ranking. CMR is source-free — it never sees the source labels at ranking time.

How CMR complements OSIRIS's Theorem 7 bound
